2. Lexical

UTF-8 Unicode. Tokens: KEYWORD IDENTIFIER LITERAL OPERATOR DELIMITER COMMENT @hw:: #[...]. ~40 keywords: fn let mut const struct enum trait impl mod module use if else for while loop match return break continue unsafe async await spawn chan select extern type where pub self Self super crate box move in as is kernel circuit tensor qubit true false.

i8..i128 u8..u128 f16/f32/f64/f128 fp16/bf16/tf32 bool char qubit[N] *mut/*const/&/&mut v128/v256/v512, struct/enum/trait, [T;N] [T] Tensor<T,Shape> stream<T> fn()->T, GpuTensor FpgaBuf, const-generics matmul<M,K,N> (mismatch = error).

7. Control flow

while i > 1 { result = result * i; i = i - 1; }
let r = match x { 1 => 10, 2 => 20, _ => 0 };
for i in 0..10 { if i % 2 == 0 { continue } s = s + i }
loop { i = i + 1; if i == 5 { break i } }
